# pi-im Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build pi-im single-package dual-channel (Feishu + Telegram) Pi extension with shared main session, pi-web holder keep-alive, and 19 IM slash commands.

**Architecture:** Core (queue/dedup/router/renderer interface) shared by two gateway adapters (Feishu Lark SDK WebSocket, Telegram Bot API long-poll); controller orchestrates auth/queue/commands; pi-bridge injects via sendUserMessage and captures message_end/tool_*/agent_settled events; extension.ts wires session_start autostart + pi-web liveness.

**Tech Stack:** TypeScript ESM, Node >=22.19, @larksuiteoapi/node-sdk ^1.72.0, Pi ExtensionAPI (@earendil-works/pi-coding-agent peer 0.83.0), vitest, biome + tsc --noEmit gate.

## Global Constraints

- Package name pi-im, pi.extensions ./dist/extension.js, target dir /Users/pan/Documents/Project/pi-im — one line each, exact values from spec.
- Config paths ~/.pi/agent/pi-im/{config.json,feishu.json,telegram.json}, mode 0600, atomic tmp+rename write.
- feishu.json {appId, appSecret, domain feishu|lark, autoStart, allowlist[]}, telegram.json {botToken, autoStart, allowlist[]}, allowlist empty = open.
- Env overrides FEISHU_APP_ID/SECRET, TELEGRAM_BOT_TOKEN, PI_IM_AUTO_START.
- No per-chat session mapping, no /conv, no multi-bot (1 per channel), no voice, no slash panel, no QR provisioning, no preset/permission commands.
- Progress: Feishu editable card 500ms/120B throttle, TG placeholder editMessage with plaintext fallback.
- Images: 30MB/single, 20/message, model copy 5MB/single 20MB total; dedup message_id 48h disk + 12h TTL FIFO, stale filter 30min.
- /batch /send /cancel DM-only plain-text <=10; /repair Feishu-DM-only, TG returns unsupported; /clear degrades to unsupported message.
- npm run check (biome + tsc --noEmit) must pass before each commit.

```ts
// src/core/queue.ts
export class SerialMessageQueue {
  private tail: Promise<void> = Promise.resolve();
  private pending = new Map<string, number>();
  position(chatId: string): number { return this.pending.get(chatId) ?? 0; }
  enqueue(chatId: string, job: () => Promise<void> | void): Promise<void> {
    this.pending.set(chatId, (this.pending.get(chatId) ?? 0) + 1);
    const run = this.tail.then(async () => { await job(); }).finally(() => {
      const n = (this.pending.get(chatId) ?? 1) - 1;
      if (n <= 0) this.pending.delete(chatId); else this.pending.set(chatId, n);
    });
    this.tail = run;
    return run;
  }
  cancelChat(chatId: string): number { const n = this.pending.get(chatId) ?? 0; this.pending.delete(chatId); return n; }
  get size(): number { let s = 0; for (const v of this.pending.values()) s += v; return s; }
}
```

```ts
export class MessageDeduplicator {
  private ids = new Map<string, number>();
  constructor(private ttlMs = 12 * 3600 * 1000, private max = 5000) {}
  seen(id: string): boolean { const t = this.ids.get(id); if (t === undefined) return false; if (Date.now() - t > this.ttlMs) { this.ids.delete(id); return false; } return true; }
  mark(id: string, now = Date.now()): void { this.ids.set(id, now); if (this.ids.size > this.max) { const first = this.ids.keys().next().value as string; this.ids.delete(first); } }
}
```

```ts
// src/core/batch.ts
export class BatchCollector {
  private active = new Set<string>();
  private items = new Map<string, string[]>();
  start(chatId: string): string { this.active.add(chatId); this.items.set(chatId, []); return "批量输入已开启,后续纯文字将暂存最多10条,/send 提交,/cancel 取消。"; }
  isActive(chatId: string): boolean { return this.active.has(chatId); }
  collect(chatId: string, text: string): string {
    if (!this.active.has(chatId)) return "未进入批量输入模式。";
    const arr = this.items.get(chatId) ?? [];
    if (arr.length >= 10) return "已收录10条,请 /send 提交。";
    arr.push(text); this.items.set(chatId, arr);
    return arr.length >= 10 ? "已收录第10条,请 /send 提交。" : `已收录第${arr.length}条。`;
  }
  submit(chatId: string): string { const arr = this.items.get(chatId) ?? []; this.active.delete(chatId); this.items.delete(chatId); return arr.join("\n"); }
  cancel(chatId: string): boolean { const had = this.active.has(chatId); this.active.delete(chatId); this.items.delete(chatId); return had; }
}
```

```ts
// src/pi-bridge.ts
export interface ProgressObserver { onText(t: string): void; onActivity(t: string): void; }
export class PiAgentBridge {
  private texts: string[] = [];
  private activities: string[] = [];
  private settled = false;
  private waiter: (() => void) | null = null;
  private cancelled = false;
  constructor(private send: (prompt: string) => Promise<void>) {}
  async run(text: string, observer: ProgressObserver): Promise<string> {
    this.texts = []; this.activities = []; this.settled = false; this.cancelled = false;
    await this.send(text);
    while (!this.settled && !this.cancelled) { await new Promise<void>((r) => { this.waiter = r; setTimeout(r, 50); }); }
    const out = [...this.activities.map((a) => `- ${a}`), ...this.texts].join("\n");
    observer.onText(out || "(空回复)");
    return out || "(空回复)";
  }
  cancel(_reason: string): void { this.cancelled = true; this.waiter?.(); }
  steer(_text: string): boolean { return false; }
  captureMessage(msg: { role?: string; content?: Array<{ type: string; text?: string }> }): void {
    if (msg.role !== "assistant") return;
    for (const c of msg.content ?? []) if (c.type === "text" && c.text) this.texts.push(c.text);
  }
  captureStreamingMessage(msg: unknown): void { this.captureMessage(msg as never); }
  captureToolStart(name: string): void { this.activities.push(`tool ${name}`); }
  captureToolEnd(): void {}
  settle(): void { this.settled = true; this.waiter?.(); }
}
```

```ts
// src/telegram/gateway.ts
import type { ChannelLike, InboundMessage } from "../core/types.js";
export function shouldRespondInGroup(_text: string, mentioned: boolean, isReplyToBot: boolean): boolean {
  return mentioned || isReplyToBot;
}
export class TelegramGateway implements ChannelLike {
  private handler: ((m: InboundMessage) => void) | null = null;
  private offset = 0;
  private stopped = false;
  constructor(private opts: { botToken: string }) { void this.opts; }
  onMessage(h: (m: InboundMessage) => void): () => void { this.handler = h; return () => { this.handler = null; }; }
  async start(): Promise<void> { this.stopped = false; void this.poll(); }
  async stop(): Promise<void> { this.stopped = true; }
  private async poll(): Promise<void> {
    while (!this.stopped) {
      try {
        const res = await fetch(`https://api.telegram.org/bot${this.opts.botToken}/getUpdates?offset=${this.offset}&timeout=30`);
        const data = (await res.json()) as { result?: Array<{ update_id: number; message?: { message_id: number; chat: { id: number; type: string }; from?: { id: number; first_name?: string }; text?: string }> } };
        for (const u of data.result ?? []) {
          this.offset = u.update_id + 1;
          const m = u.message; if (!m?.text) continue;
          this.handler?.({ messageId: `tg-${u.update_id}`, chatId: String(m.chat.id), chatType: m.chat.type === "private" ? "p2p" : "group", senderId: String(m.from?.id ?? 0), senderName: m.from?.first_name, text: m.text, mentionBot: m.text.includes("@"), timestampMs: Date.now() });
        }
      } catch { await new Promise((r) => setTimeout(r, 3000)); }
    }
  }
  async sendText(chatId: string, text: string): Promise<string> {
    const res = await fetch(`https://api.telegram.org/bot${this.opts.botToken}/sendMessage`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: chatId, text: text.slice(0, 4000) }) });
    const data = (await res.json()) as { result?: { message_id: number } };
    return String(data.result?.message_id ?? Date.now());
  }
  async editText(chatId: string, messageId: string, text: string): Promise<void> {
    await fetch(`https://api.telegram.org/bot${this.opts.botToken}/editMessageText`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: chatId, message_id: Number(messageId) || undefined, text: text.slice(0, 4000) }) });
  }
}
```

```ts
export interface ParsedRemoteCommand { name: string; args: string; }
const KNOWN = ["/help","/status","/version","/new","/stop","/steer","/compact","/history","/models","/model","/thinking","/reasoning","/workspace","/ws","/wsl","/sessionlist","/sessions","/session","/batch","/send","/cancel","/repair","/clear"];
export function parseRemoteCommand(input: string): ParsedRemoteCommand | undefined {
  const t = input.trim(); if (!t.startsWith("/")) return undefined;
  const i = t.search(/\s/); const name = (i === -1 ? t : t.slice(0, i)).toLowerCase();
  if (!KNOWN.includes(name)) return { name, args: i === -1 ? "" : t.slice(i + 1).trim() };
  return { name, args: i === -1 ? "" : t.slice(i + 1).trim() };
}
export function renderHelp(): string {
  return ["/help /status /version /new /stop /steer <文本> /compact /history [N] /models /model [序号|ID] /thinking [ID] (/reasoning) /workspace <路径|序号> (/ws) /wsl /sessionlist (/sessions) /session <ID|序号> /batch /send /cancel /repair /clear"].join("\n");
```

```ts
  constructor(private channel: ChannelLike, private agent: PiAgentBridge, private opts: { allowlist: string[]; channelName: string }) {}
  private allowed(senderId: string): boolean { if (this.opts.allowlist.length === 0) return true; return this.opts.allowlist.includes(senderId); }
  async handleInbound(m: InboundMessage): Promise<void> {
    if (this.dedup.seen(m.messageId)) return;
    this.dedup.mark(m.messageId);
    if (Date.now() - m.timestampMs > 30 * 60 * 1000) return;
    if (!this.allowed(m.senderId)) return;
    if (m.chatType === "group" && !m.mentionBot) return;
    const cmd = parseRemoteCommand(m.text);
    if (cmd) { await this.handleCommand(m, cmd.name, cmd.args); return; }
    if (this.batch.isActive(m.chatId)) { await this.channel.sendText(m.chatId, this.batch.collect(m.chatId, m.text)); return; }
    const pos = this.queue.size + 1;
    let progressId = "";
    await this.queue.enqueue(m.chatId, async () => {
      progressId = await this.channel.sendText(m.chatId, pos > 1 ? `排队第${pos}位…` : "收到,处理中…");
      const out = await this.agent.run(m.text, { onText: () => {}, onActivity: () => {} });
      await this.channel.editText(m.chatId, progressId, out.slice(0, 4000)).catch(() => this.channel.sendText(m.chatId, out.slice(0, 4000)));
    });
  }
  private async handleCommand(m: InboundMessage, name: string, args: string): Promise<void> {
    switch (name) {
      case "/help": await this.channel.sendText(m.chatId, renderHelp()); return;
      case "/version": await this.channel.sendText(m.chatId, "pi-im 0.1.0"); return;
      case "/batch": if (m.chatType !== "p2p") { await this.channel.sendText(m.chatId, "仅私聊可用。"); return; } await this.channel.sendText(m.chatId, this.batch.start(m.chatId)); return;
      case "/send": { const all = this.batch.submit(m.chatId); await this.handleInbound({ ...m, text: all || "(空批量)", messageId: `${m.messageId}-send` }); return; }
      case "/cancel": await this.channel.sendText(m.chatId, this.batch.cancel(m.chatId) ? "已取消。" : "无批量任务。"); return;
      case "/repair": await this.channel.sendText(m.chatId, this.opts.channelName === "telegram" ? "TG 不支持 /repair。" : "请在设置页补全 scope 后重试。"); return;
      default: await this.channel.sendText(m.chatId, `命令 ${name} ${args} 已收到(V1最小实现)。`); return;
    }
  }
}
```
